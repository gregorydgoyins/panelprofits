// Helper for HTTP HEAD verification
async function verifyUrl(url: string): Promise<boolean> {

    const res = await fetch(url, { method: "HEAD", signal: AbortSignal.timeout(3500) });
    return res.ok;
  } catch {
    return false;
  }

async function main() {
  console.log("=== BUILDING VERIFIED MULTI-EXCHANGE AUCTIONS CATALOG ===");

  const allListings = [
    ...MYCOMICSHOP_LIVE_AUCTIONS,
    ...MERCARI_LIVE_LISTINGS,
    ...VERIFIED_EXCHANGE_LISTINGS
  ];

  console.log(`Total listings to verify: ${allListings.length}`);

  const verified: RawAuctionListing[] = [];
  let failCount = 0;

  for (const item of allListings) {
    const isOk = await verifyUrl(item.imageUrl);
    if (isOk) {
      verified.push(item);
      console.log(`[PASS] ${item.source.toUpperCase().padEnd(12)}: ${item.title.substring(0, 50)}`);
    } else {
      failCount++;
      console.error(`[FAIL] ${item.source.toUpperCase().padEnd(12)}: ${item.title} -> URL: ${item.imageUrl}`);
    }
  }

  console.log(`\nVerification Summary: ${verified.length} passed, ${failCount} failed.`);

  if (failCount === 0) {
    const outPath = path.resolve(process.cwd(), "data", "live_auctions.json");
    fs.writeFileSync(outPath, JSON.stringify(verified, null, 2), "utf-8");
    console.log(`Successfully wrote ${verified.length} 100% verified live auction listings to ${outPath}`);
  } else {
    console.error("FATAL: Cannot write live_auctions.json with broken image URLs!");
    process.exit(1);
  }
}

main().catch(err => {
  console.error("Error building live auctions:", err);
  process.exit(1);
});
