/**
 * Interleave assets so that consecutive cards exhibit visual and category diversity.
 */
function interleaveInitialAssets(items: AssetItem[]): AssetItem[] {
  if (items.length <= 1) return [...items];

  const remaining = [...items];
  const result: AssetItem[] = [];

  const firstIdx = remaining.findIndex((it) => it.coverImageUrl && it.coverImageUrl !== "");
  result.push(remaining.splice(firstIdx >= 0 ? firstIdx : 0, 1)[0]);

  while (remaining.length > 0) {
    const lastItem = result[result.length - 1];

    let nextIdx = remaining.findIndex(
      (it) =>
        it.coverImageUrl !== lastItem.coverImageUrl &&
        it.assetType !== lastItem.assetType
    );

    if (nextIdx === -1) {
      nextIdx = remaining.findIndex((it) => it.coverImageUrl !== lastItem.coverImageUrl);
    }

    if (nextIdx === -1) {
      for (let i = 0; i < remaining.length; i++) {
        if (!lastItem || remaining[i].coverImageUrl !== lastItem.coverImageUrl) {
          nextIdx = i;
          break;
        }
      }
    }

    if (nextIdx === -1) nextIdx = 0;
    result.push(remaining.splice(nextIdx, 1)[0]);
  }

  // Ensure track wrap-around boundary doesn't match first item
  if (result.length > 2 && result[result.length - 1].coverImageUrl === result[0].coverImageUrl) {
    const last = result.pop()!;
    result.splice(Math.floor(result.length / 2), 0, last);
  }

  return result;
}

export const INITIAL_SURFACE_ASSETS: AssetItem[] = interleaveInitialAssets(RAW_INITIAL_SURFACE_ASSETS);
